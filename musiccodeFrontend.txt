const MusicScreen = () => {
  const [songs, setSongs] = useState(initialSongs);
  const [playingId, setPlayingId] = useState(null);
  const soundRef = useRef(null);

  // Stop any currently playing sound
  const stopCurrentSound = () => {
    if (soundRef.current) {
      soundRef.current.stop();
      soundRef.current.release();
      soundRef.current = null;
    }
  };

  const playSong = (song) => {
    // If clicking the same song that's currently playing, stop it
    if (playingId === song.id) {
      stopCurrentSound();
      setPlayingId(null);
      return;
    }

    // Stop any currently playing sound before playing a new one
    stopCurrentSound();

    // Start playing the new song
    const streamUrl = `${BASE_URL}/audio/play/${song.filename}`;

    const newSound = new Sound(streamUrl, null, (error) => {
      if (error) {
        console.log('Failed to load sound', error);
        return;
      }
      
      newSound.play((success) => {
        if (success) {
          console.log('Finished playing');
        } else {
          console.log('Playback failed');
        }
        // Only release and reset if this is still the current sound
        if (soundRef.current === newSound) {
          soundRef.current.release();
          soundRef.current = null;
          setPlayingId(null);
        }
      });
    });

    soundRef.current = newSound;
    setPlayingId(song.id);
  };

  // Clean up on component unmount
  useEffect(() => {
    return () => {
      stopCurrentSound();
    };
  }, []);

  return (
    <>
      <View style={styles.container}>
        <Text style={styles.header}>Playlists</Text>
        <Image source={image1} style={styles.playimage} />
      </View>

      <View style={styles.playlistContainer}>
        <View style={styles.play1}>
          <Image source={image1} style={styles.playlistImage} />
          <View style={styles.playlistInfo}>
            <View style={styles.playtext}>
              <Text style={styles.playlistTitle}>Lo-Fi Beats</Text>
              <Text style={styles.playlistDescription}>Just Relax and Listen</Text>
              <View style={styles.chilldiv}>
                <Image source={image1} style={styles.chillimage} />
                <Text style={styles.playlistArtist}>Chill Beats Music</Text>
              </View>
            </View>
            <View style={styles.playsong}>
              <Image source={image1} style={styles.playimage2} />
            </View>
          </View>
        </View>
        <View style={styles.rectangle} />
      </View>

      <ScrollView style={styles.songContainer}>
        <Text style={styles.songHeader}>Songs</Text>
        {songs.map((song) => (
          <TouchableOpacity
            key={song.id}
            style={[styles.card, playingId === song.id && styles.playing]}
            onPress={() => playSong(song)}
          >
            <View style={styles.songInfo}>
              <View style={styles.textContainer}>
                <Image source={song.image} style={styles.songImage} />
                <View>
                  <Text style={styles.songTitle}>{song.title}</Text>
                  <Text style={styles.songArtist}>{song.artist}</Text>
                </View>
              </View>
              <View style={styles.playIconContainer}>
                <Image source={image1} style={styles.playIcon} />
              </View>
            </View>
          </TouchableOpacity>
        ))}
      </ScrollView>
    </>
  );
};
